```
apiVersion: apps/v1
kind: StatefulSet
metadata:
  name: mysql
spec:
  serviceName: "mysql"          # 必须配一个 Headless Service
  replicas: 3
  selector:
    matchLabels:
      app: mysql
  template:
    metadata:
      labels:
        app: mysql
    spec:
      containers:
      - name: mysql
        image: mysql:5.7
        env:
        - name: MYSQL_ROOT_PASSWORD
          value: "123456"
        volumeMounts:
        - name: data              # 这里的 name 对应下面模板的 name
          mountPath: /var/lib/mysql
  volumeClaimTemplates:           # 关键：每个 Pod 自动生成独立 PVC
  - metadata:
      name: data
    spec:
      accessModes: ["ReadWriteOnce"]
      storageClassName: "nfs-client"   # 换成你的 StorageClass，如 csi-rbd-sc
      resources:
        requests:
          storage: 10Gi
```
